"""Drive the host-compiled firmware glue through the project's real Python transport over TCP.

Mode "pyserial": serial.serial_for_url("socket://...") as is. Mode "adapter": a thin pyserial-like wrapper whose
in_waiting reports "many" when readable (pyserial's socket:// in_waiting returns only 0/1)."""
import select
import socket
import sys
import time

import serial

from autopark.transport.vehicle_io import SerialVehicleIO

port = int(sys.argv[1]) if len(sys.argv) > 1 else 5555
mode = sys.argv[2] if len(sys.argv) > 2 else "adapter"


class SocketTransport:
    def __init__(self, host, port, timeout=0.01):
        self.s = socket.create_connection((host, port))
        self.s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
        self.timeout = timeout

    @property
    def in_waiting(self):
        r, _, _ = select.select([self.s], [], [], 0)
        return 4096 if r else 0

    def read(self, n):
        r, _, _ = select.select([self.s], [], [], self.timeout)
        if not r:
            return b""
        data = self.s.recv(n)
        if not data:
            raise OSError("socket closed")
        return data

    def write(self, data):
        self.s.sendall(data)

    def close(self):
        self.s.close()


t = SocketTransport("127.0.0.1", port) if mode == "adapter" else serial.serial_for_u
